window.handleDeposit = (userID, rid, amount) => { // یہاں ہم نے کسٹمر کی لکھی ہوئی رقم (مثلاً 2) کو 300 سے ضرب دے دی // اب اگر کسٹمر 2 لکھے گا تو اسے 600 روپے ملیں گے const pkrAmount = parseFloat(amount) * 300; const userRef = ref(db, `users/${userID}`); // بیلنس کو روپے (PKR) میں اپ ڈیٹ کرنا runTransaction(ref(db, `users/${userID}/balance`), (current) => { return (parseFloat(current) || 0) + pkrAmount; }).then(() => { // ٹوٹل ڈپازٹ کو بھی روپے میں اپ ڈیٹ کریں get(userRef).then((snap) => { if (snap.exists()) { const curDep = snap.val().totalDeposit || 0; update(userRef, { totalDeposit: (parseFloat(curDep) + pkrAmount) }); } }); // ڈپازٹ ریکویسٹ کو اپروو کر دیں update(ref(db, `deposits/${rid}`), { status: 'Approved' }); alert("کامیابی! $" + amount + " یعنی Rs. " + pkrAmount + " یوزر کے اکاؤنٹ میں ایڈ ہو گئے ہیں۔"); }).catch(err => alert("Error: " + err.message)); };
